# flag: r
# Pacote de redes: exemplos comuns (vale pra qualquer máquina).
# Formato: pedido => comando. Os exemplos desta máquina ficam em ~/.config/fa/pacotes/redes.txt.
mostra uma amostra dos pacotes capturados => sudo tcpdump -i <interface ativa> -c 20 -nn
captura só o tráfego dns => sudo tcpdump -i <interface ativa> -nn port 53
captura o tráfego de um ip => sudo tcpdump -i <interface ativa> -nn host 192.168.1.10
salva a captura num arquivo => sudo tcpdump -i <interface ativa> -w captura.pcap
lê uma captura salva => tcpdump -nn -r captura.pcap | head -50
mostra as interfaces e ips => ip -br addr
qual o meu ip => ip -br addr show <interface ativa>
qual o meu ip público => curl -s https://api.ipify.org; echo
qual o gateway => ip route show default
mostra a tabela de rotas => ip route
por onde sai o pacote para um ip => ip route get 8.8.8.8
mostra a tabela arp => ip neigh
quais portas estão escutando => sudo ss -tulnp
quem está conectado nesta máquina => ss -tnp state established
qual programa usa a porta 8080 => sudo ss -tlnp 'sport = :8080'
testa se uma porta remota está aberta => nc -vz -w 5 host 443
pinga um host 4 vezes => ping -c 4 host
mostra o caminho até um host => tracepath host
mede perda e latência no caminho => mtr -rwc 20 host
resolve um nome => dig +short nome
resolve com um dns específico => dig @1.1.1.1 +short nome
qual dns estou usando => resolvectl dns
mostra os cabeçalhos http de um site => curl -sI https://site
quanto tempo leva cada etapa de uma requisição => curl -o /dev/null -s -w 'dns %{time_namelookup}s conexão %{time_connect}s tls %{time_appconnect}s total %{time_total}s\n' https://site
descobre os hosts ativos na rede => sudo nmap -sn 192.168.1.0/24
mostra as redes wifi => nmcli device wifi list
mostra as conexões do networkmanager => nmcli connection show
liga uma conexão => nmcli connection up <nome>
tráfego por interface agora => ip -s link
